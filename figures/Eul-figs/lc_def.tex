{\footnotesize
% panel: #1..#4 = bases at v,a,b,c ; #5 = edges ; #6 = panel caption
\newcommand{\lcp}[6]{%
  \begin{tikzpicture}[baseline={(0,0)},x=1cm,y=1cm,
      gv/.append style={minimum size=6mm,font=\footnotesize},
      dele/.style={Xcol,line width=1.3pt,dashed}]
    \node[gv,frv,label={[label distance=-1pt]below:$#1$}] (v) at (0,0) {$4$};
    \node[gv,label={[label distance=-1pt]below:$#2$}] (a) at (-1.2,-0.75) {$1$};
    \node[gv,label={[label distance=-1pt]below:$#3$}] (b) at (1.2,-0.75) {$2$};
    \node[gv,label={[label distance=-1pt]above:$#4$}] (c) at (0,1.05) {$3$};
    \begin{pgfonlayer}{background}
    \draw[ge](v)--(a);\draw[ge](v)--(b);\draw[ge](v)--(c);
    #5
    \end{pgfonlayer}
    \node[text=Vcol,align=center] at (0,-1.6) {#6};
  \end{tikzpicture}}
