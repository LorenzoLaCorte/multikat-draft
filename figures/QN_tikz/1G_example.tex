\begin{tikzpicture}[
    % ── Tunable parameters ─────────────────────────────────────────
    nodesize/.store in   = \nodesize,   nodesize   = 0.8cm,
    nodethick/.store in  = \nodethick,  nodethick  = thick,
    nodesep/.store in    = \nodesep,    nodesep    = 2.5cm,
    linkthick/.store in  = \linkthick,  linkthick  = thick,
    pairbend/.store in   = \pairbend,   pairbend   = 20,
    swapbend/.store in   = \swapbend,   swapbend   = 16,
    finalbend/.store in  = \finalbend,  finalbend  = 10,
    pairthick/.store in  = \pairthick,  pairthick  = thick,
    rowsep/.store in     = \rowsep,     rowsep     = 1.4cm,
    % ── Derived styles ─────────────────────────────────────────────
    node/.style={
        circle, draw, \nodethick,
        minimum size=\nodesize,
        inner sep=0pt
    },
    ep/.style={dashed, \pairthick, bend left=\pairbend},
    swapep/.style={dashed, \pairthick, bend left=\swapbend},
    finalep/.style={dashed, \pairthick, bend left=\finalbend},
]

% Color scheme:
%   swapB color  = blue    (A~B and B~C consumed at B)
%   swapD color  = orange  (C~D and D~E consumed at D)
%   swapC color  = violet  (A~C and C~E consumed at C)
%   final color  = red     (A~E, end-to-end)

% ════════════════════════════════════════════════════════════════════
% ROW 0 — Physical topology
% ════════════════════════════════════════════════════════════════════
\foreach \lbl/\i in {A/0, B/1, C/2, D/3, E/4}{
    \node[node] (\lbl0) at (\i*\nodesep, 0) {$\lbl$};
}
\foreach \u/\v in {A0/B0, B0/C0, C0/D0, D0/E0}{
    \draw[\linkthick] (\u) -- (\v);
}
\node[left=0.2cm of A0] {\small Step 1};

% ════════════════════════════════════════════════════════════════════
% ROW 1 — Elementary link generation
% ════════════════════════════════════════════════════════════════════
\foreach \lbl/\i in {A/0, B/1, C/2, D/3, E/4}{
    \node[node] (\lbl1) at (\i*\nodesep, -\rowsep) {$\lbl$};
}
\foreach \u/\v in {A1/B1, B1/C1, C1/D1, D1/E1}{
    \draw[\linkthick] (\u) -- (\v);
}
% A~B and B~C: both consumed at swap B → blue
\draw[blue,   ep] (A1) to (B1);
\draw[blue,   ep] (B1) to (C1);
% C~D and D~E: both consumed at swap D → orange
\draw[orange, ep] (C1) to (D1);
\draw[orange, ep] (D1) to (E1);
\node[left=0.2cm of A1] {\small Step 2};

% ════════════════════════════════════════════════════════════════════
% ROW 2 — Parallel swaps at B and D
% ════════════════════════════════════════════════════════════════════
\foreach \lbl/\i in {A/0, B/1, C/2, D/3, E/4}{
    \node[node] (\lbl2) at (\i*\nodesep, -2*\rowsep) {$\lbl$};
}
\foreach \u/\v in {A2/B2, B2/C2, C2/D2, D2/E2}{
    \draw[\linkthick] (\u) -- (\v);
}
% consumed elementary pairs (faded)
\draw[blue,   ep, opacity=0.15] (A2) to (B2);
\draw[blue,   ep, opacity=0.15] (B2) to (C2);
\draw[orange, ep, opacity=0.15] (C2) to (D2);
\draw[orange, ep, opacity=0.15] (D2) to (E2);
% A~C and C~E: both consumed at final swap C → violet
\draw[violet, swapep] (A2) to (C2);
\draw[violet, swapep] (C2) to (E2);
% swap markers
\node[above=-0.05cm of B2, font=\footnotesize, blue]   {swap};
\node[above=-0.05cm of D2, font=\footnotesize, orange] {swap};
\node[left=0.2cm of A2] {\small Step 3};

% ════════════════════════════════════════════════════════════════════
% ROW 3 — Final swap at C
% ════════════════════════════════════════════════════════════════════
\foreach \lbl/\i in {A/0, B/1, C/2, D/3, E/4}{
    \node[node] (\lbl3) at (\i*\nodesep, -3*\rowsep) {$\lbl$};
}
\foreach \u/\v in {A3/B3, B3/C3, C3/D3, D3/E3}{
    \draw[\linkthick] (\u) -- (\v);
}

% final end-to-end pair
\draw[red, finalep] (A3) to (E3);
\node[above=0.06cm of C3, font=\footnotesize, violet] {swap};
\node[left=0.2cm of A3] {\small Step 4};

% ════════════════════════════════════════════════════════════════════
% Step descriptions (right margin)
% ════════════════════════════════════════════════════════════════════
\node[right=0.3cm of E0, align=left, font=\small]
    {Physical topology};
\node[right=0.3cm of E1, align=left, font=\small]
    {Elementary link\\generation};
\node[right=0.3cm of E2, align=left, font=\small]
    {Parallel swaps\\at $B$, $D$};
\node[right=0.3cm of E3, align=left, font=\small]
    {Final swap at $C$\\$\Rightarrow$ end-to-end $A$\,--\,$E$};

\end{tikzpicture}
