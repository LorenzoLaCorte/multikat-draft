% Quantum key distribution, told with shapes only:
%   two nodes share entanglement (blue) -> both end up with the SAME key
%   (gold); anyone listening on the channel is locked out (crossed-out eye).
% Palette from sections/macros-eulvec.tex.  Every line ends in % so \input'ing
% this inside a \resizebox adds no stray space.
\begin{tikzpicture}[%
  nbox/.style = {rounded corners=3pt, draw=Vcol, line width=1.1pt,%
                 fill=Vcol!8, minimum width=16mm, minimum height=16mm},%
  qb/.style   = {circle, draw=Vcol!55, fill=Vcol!18, minimum size=4.5mm,%
                 inner sep=0pt, line width=.6pt},%
  link/.style = {Zcol, line width=1.6pt},%
  key/.style  = {Acc, line width=1.4pt},%
  eye/.style  = {Vcol!55, line width=1.1pt},%
  no/.style   = {Xcol, line width=1.6pt, line cap=round},%
]%
  \useasboundingbox (-0.95,-2.25) rectangle (5.95,2.05);%
  % the two parties
  \node[nbox] (L) at (0,0) {};  \node[qb] at (0,0) {};%
  \node[nbox] (R) at (5,0) {};  \node[qb] at (5,0) {};%
  \onslide<1>{\draw[link] (L) -- (R);}%
  % the eavesdropper on the channel: an eye, crossed out
  \begin{scope}[shift={(2.5,1.15)}, scale=1.25]%
    \draw[eye] (-0.55,0) .. controls (-0.25,0.45) and (0.25,0.45) .. (0.55,0)%
               .. controls (0.25,-0.45) and (-0.25,-0.45) .. (-0.55,0);%
    \fill[Vcol!55] (0,0) circle (0.16);%
    \draw[no] (-0.62,-0.5) -- (0.62,0.5);%
  \end{scope}%
  % the same key on both sides
  \def\keyat#1{\begin{scope}[shift={(#1,-1.5)}, scale=1.3]%
      \draw[key] (0,0) circle (0.22);%
      \draw[key] (0.22,0) -- (0.95,0);%
      \draw[key] (0.72,0) -- (0.72,-0.26);%
      \draw[key] (0.93,0) -- (0.93,-0.26);%
    \end{scope}}%
  \onslide<2->{\keyat{-0.45}\keyat{4.55}}%
\end{tikzpicture}%
