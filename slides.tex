\begin{columns}[c]

  \end{scope}

  \draw[bigarr,usiblue] (.5,-.45) -- node[left,text=usiblue] {$\pi$} (.5,-1.05);

\[ \pi\big(\localComp v(G)\big) = \localComp{\pi(v)}\big(\pi(G)\big) \]

\cl{Any mix of the two reduces to local complementations followed by one relabelling}
\end{column}
\end{columns}
\end{frame}
